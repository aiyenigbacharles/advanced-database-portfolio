SELECT o.order_id, o.customer_id, o.total_amount
FROM orders o
JOIN (SELECT customer_id, AVG(total_amount) AS avg_total
      FROM orders GROUP BY customer_id) a ON a.customer_id = o.customer_id
WHERE o.total_amount > a.avg_total;
